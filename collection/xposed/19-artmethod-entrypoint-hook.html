ART ArtMethod 入口替换 Hook 原理

概述

前面几篇已经讲过 Java Hook、Native Hook、 RegisterNatives 监控和 libxposed 新 API。这一篇往更底层走一步：直接看 ART 运行时里的 ArtMethod ，用修改方法入口的方式，把一个 Java 方法重定向到另一个 Java 方法。

本文环境以 Android SDK 28（Android 9 / Pie） 为准，示例只讨论 64 位 arm64 和 x86_64 下的实验性实现。 ArtMethod 是 ART 内部结构，不属于稳定公开 API，跨系统版本复用时必须重新确认字段布局。

示例目标很小：

public fun add(x: Int, y: Int): Int {
    return x + y
}

public fun addHook(x: Int, y: Int): Int {
    return x * y
}

initHook() 执行后，后续调用 add(10, 20) 会进入 addHook(10, 20) ，返回值从 30 变成 200 。

核心链路如下：

Java Method 对象
  ↓ 读取 java.lang.reflect.Executable.artMethod
ArtMethod*
  ↓ 修改 ptr_sized_fields_.entry_point_from_quick_compiled_code_
目标方法 quick entry
  ↓
自定义 trampoline
  ↓
addHook 的 quick entry

这不是一个完整的 Xposed 框架实现，而是一个很好的实验样本：它把 “方法为什么能被替换” 这件事拆到 ART 方法结构、quick entry 和机器码跳板这一层。

一、最小复现代码

为了让这篇文章不只停在原理层，先给出一套最小代码。建议使用 Android 9 / API 28 的 64 位模拟器或真机，先用 debug 构建验证，不开启混淆和压缩。

复现时只需要观察一件事：

Hook 前：add(10, 20) = 30
Hook 后：add(10, 20) = 200

2. Activity 完整代码

界面只放一个 TextView ，点击第一次安装 Hook，随后调用 add(10, 20) 。为了方便对比，点击前会先打印一次原始结果。

package com.example.artentry

import android.app.Activity
import android.os.Bundle
import android.util.Log
import android.widget.TextView

class MainActivity : Activity() {

    private var hookInstalled = false
    private lateinit var output: TextView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        output = TextView(this).apply {
            textSize = 18f
            text = "Native: ${stringFromJNI()}\nClick to install ArtMethod hook"
            setPadding(48, 96, 48, 48)
        }
        setContentView(output)

        Log.d(TAG, "before hook add(10, 20): ${add(10, 20)}")

        output.setOnClickListener {
            if (!hookInstalled) {
                hookInstalled = initHook()
                Log.d(TAG, "initHook: $hookInstalled")
            }

            val result = add(10, 20)
            Log.d(TAG, "after hook add(10, 20): $result")
            output.text = "hookInstalled=$hookInstalled\nadd(10, 20)=$result"
        }
    }

    external fun stringFromJNI(): String

    external fun initHook(): Boolean

    fun add(x: Int, y: Int): Int {
        return x + y
    }

    fun addHook(x: Int, y: Int): Int {
        return x * y
    }

    companion object {
        private const val TAG = "ArtEntry"

        init {
            System.loadLibrary("artentry")
        }
    }
}

3. ArtMethod 结构定义

Android SDK 28 的 64 位 ART 中，本文用到的关键偏移是 entry_point_from_quick_compiled_code_ = 0x20 。下面只保留复现所需字段：

#ifndef ART_ENTRY_ART_METHOD_H
#define ART_ENTRY_ART_METHOD_H

#include <atomic>
#include <cstdint>

class ArtMethod final {
public:
    uint32_t declaring_class_;

    std::atomic<std::uint32_t> access_flags_;

    uint32_t dex_code_item_offset_;

    uint32_t dex_method_index_;

    uint16_t method_index_;

    uint16_t hotness_count_;

    struct PtrSizedFields {
        void *data_;

        void *entry_point_from_quick_compiled_code_;
    } ptr_sized_fields_;
};

#endif

4. Native Hook 完整代码

这段代码完成了从 Method 到 ArtMethod* ，再到 quick entry 替换的完整流程。包名如果改掉， Java_com_example_artentry_MainActivity_* 这两个 JNI 导出函数名也要同步调整。

#include <jni.h>
#include <string>
#include <sys/mman.h>
#include <android/log.h>
#include <cerrno>
#include <cstring>
#include "art_method.h"

#define TRAMPOLINE_SPACE_SIZE 4096

#define LOGD(...) __android_log_print(ANDROID_LOG_DEBUG, "ART_ENTRY", __VA_ARGS__)
#define LOGE(...) __android_log_print(ANDROID_LOG_ERROR, "ART_ENTRY", __VA_ARGS__)

static bool clearPendingException(JNIEnv *env, const char *message) {
    if (!env->ExceptionCheck()) {
        return false;
    }

    env->ExceptionDescribe();
    env->ExceptionClear();
    LOGE("%s", message);
    return true;
}

static void *allocTrampolineSpace() {
    auto *buf = static_cast<unsigned char *>(mmap(
            nullptr,
            TRAMPOLINE_SPACE_SIZE,
            PROT_READ | PROT_WRITE | PROT_EXEC,
            MAP_ANONYMOUS | MAP_PRIVATE,
            -1,
            0));

    if (buf == MAP_FAILED) {
        LOGE("mmap failed, errno = %s", strerror(errno));
        return nullptr;
    }

    LOGD("allocating space for trampoline code at %p", buf);
    return buf;
}

extern "C"
JNIEXPORT jstring JNICALL
Java_com_example_artentry_MainActivity_stringFromJNI(JNIEnv *env, jobject /* thiz */) {
    std::string hello = "Hello from C++";
    return env->NewStringUTF(hello.c_str());
}

extern "C"
JNIEXPORT jboolean JNICALL
Java_com_example_artentry_MainActivity_initHook(JNIEnv *env, jobject thiz) {
    jclass class_executable = env->FindClass("java/lang/reflect/Executable");
    if (class_executable == nullptr ||
        clearPendingException(env, "FindClass(java/lang/reflect/Executable) failed")) {
        return JNI_FALSE;
    }

    jfieldID field_art_method = env->GetFieldID(class_executable, "artMethod", "J");
    if (field_art_method == nullptr ||
        clearPendingException(env, "GetFieldID(artMethod) failed")) {
        return JNI_FALSE;
    }

    jclass obj_class = env->GetObjectClass(thiz);
    if (obj_class == nullptr || clearPendingException(env, "GetObjectClass(thiz) failed")) {
        return JNI_FALSE;
    }

    jmethodID method_add = env->GetMethodID(obj_class, "add", "(II)I");
    jmethodID method_add_hook = env->GetMethodID(obj_class, "addHook", "(II)I");
    if (method_add == nullptr || method_add_hook == nullptr ||
        clearPendingException(env, "GetMethodID(add/addHook) failed")) {
        return JNI_FALSE;
    }

    jobject reflected_add = env->ToReflectedMethod(obj_class, method_add, JNI_FALSE);
    jobject reflected_add_hook = env->ToReflectedMethod(obj_class, method_add_hook, JNI_FALSE);
    if (reflected_add == nullptr || reflected_add_hook == nullptr ||
        clearPendingException(env, "ToReflectedMethod(add/addHook) failed")) {
        return JNI_FALSE;
    }

    auto *art_method_add = reinterpret_cast<ArtMethod *>(
            env->GetLongField(reflected_add, field_art_method));
    auto *art_method_add_hook = reinterpret_cast<ArtMethod *>(
            env->GetLongField(reflected_add_hook, field_art_method));

    if (clearPendingException(env, "GetLongField(artMethod) failed")) {
        return JNI_FALSE;
    }

    if (art_method_add == nullptr || art_method_add_hook == nullptr) {
        LOGE("artMethod is null");
        return JNI_FALSE;
    }

    LOGD("add ArtMethod: %p", art_method_add);
    LOGD("addHook ArtMethod: %p", art_method_add_hook);
    LOGD("add entry before: %p",
         art_method_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_);
    LOGD("addHook entry: %p",
         art_method_add_hook->ptr_sized_fields_.entry_point_from_quick_compiled_code_);

#if !defined(__aarch64__) && !defined(__x86_64__)
    LOGE("Unsupported ABI for trampoline");
    return JNI_FALSE;
#else
    auto *trampoline_space = static_cast<unsigned char *>(allocTrampolineSpace());
    if (trampoline_space == nullptr) {
        return JNI_FALSE;
    }

    unsigned char *target_addr = trampoline_space;

#if defined(__aarch64__)
    unsigned char trampoline[] = {
            // OatQuickMethodHeader::code_size_.
            0x00, 0x00, 0x00, 0x00,
            // ldr x0, #12; ldr x17, [x0, #32]; br x17
            0x60, 0x00, 0x00, 0x58,
            0x11, 0x10, 0x40, 0xf9,
            0x20, 0x02, 0x1f, 0xd6,
            // ArtMethod* of addHook.
            0x00, 0x00, 0x00, 0x00,
            0x00, 0x00, 0x00, 0x00,
    };
    memcpy(trampoline + 16, &art_method_add_hook, sizeof(void *));
#else
    unsigned char trampoline[] = {
            // OatQuickMethodHeader::code_size_.
            0x00, 0x00, 0x00, 0x00,
            // mov rdi, ArtMethod*; push [rdi + 0x20]; ret
            0x48, 0xbf, 0x00, 0x00, 0x00, 0x00,
            0x00, 0x00, 0x00, 0x00,
            0xff, 0x77, 0x20,
            0xc3
    };
    memcpy(trampoline + 6, &art_method_add_hook, sizeof(void *));
#endif

    memcpy(target_addr, trampoline, sizeof(trampoline));
    __builtin___clear_cache(
            reinterpret_cast<char *>(target_addr),
            reinterpret_cast<char *>(target_addr + sizeof(trampoline)));

    target_addr += 4;
    art_method_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_ = target_addr;

    uint32_t access_flags = art_method_add->access_flags_.load();
    access_flags |= 0x100;
    art_method_add->access_flags_.store(access_flags);

    LOGD("add entry after: %p",
         art_method_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_);
    LOGD("Hook success!");
    return JNI_TRUE;
#endif
}

5. 运行验证

安装到 Android 9 / API 28 的 64 位环境后，第一次启动会看到原始结果：

D/ArtEntry: before hook add(10, 20): 30

点击文本安装 Hook 后，应该看到：

D/ART_ENTRY: add ArtMethod: 0x...
D/ART_ENTRY: addHook ArtMethod: 0x...
D/ART_ENTRY: add entry before: 0x...
D/ART_ENTRY: addHook entry: 0x...
D/ART_ENTRY: allocating space for trampoline code at 0x...
D/ART_ENTRY: add entry after: 0x...
D/ART_ENTRY: Hook success!
D/ArtEntry: initHook: true
D/ArtEntry: after hook add(10, 20): 200

如果 initHook: false ，优先看 ART_ENTRY 的错误日志。最常见的问题是 ABI 不对、运行环境不是 API 28、JNI 方法签名和 Kotlin 方法签名不一致，或者包名改了但 JNI 导出函数名没有同步调整。

这个 JNI 函数做了 4 件事：

  1. 找到 add 和 addHook 的 jmethodID 。
  2. 把 jmethodID 转成 Java 反射 Method 对象。
  3. 从 Method.artMethod 读出 ART 内部的 ArtMethod* 。
  4. 修改 add 的 quick compiled code 入口，让它跳到 addHook 。

二、ArtMethod 是什么

Android Runtime 执行 Java 方法时，不只看 Dex 里的 method_id 。类加载、验证、解释执行、JIT/AOT 编译之后，运行时会有一份内部方法对象，也就是 ArtMethod 。

示例里按 Android SDK 28 的 ART 布局定义了一个简化版结构：

class ArtMethod final {
public:
    uint32_t declaring_class_;
    std::atomic<std::uint32_t> access_flags_;
    uint32_t dex_code_item_offset_;
    uint32_t dex_method_index_;
    uint16_t method_index_;
    uint16_t hotness_count_;

    struct PtrSizedFields {
        void *data_;
        void *entry_point_from_quick_compiled_code_;
    } ptr_sized_fields_;
};

先抓住几个字段：

字段含义
declaring_class_声明这个方法的 Class，64 位 ART 中通常是压缩引用
access_flags_方法访问标志，例如 public、static、native 等
dex_code_item_offset_Dex code_item 偏移
dex_method_index_Dex method_ids 里的方法索引
method_index_direct methods、vtable 或 ifTable 中的索引
hotness_count_JIT 热度计数相关字段
ptr_sized_fields_.data_不同方法类型下含义不同，native 方法常用于 JNI 函数指针等
ptr_sized_fields_.entry_point_from_quick_compiled_code_quick compiled code 调用入口

这篇文章最关心最后一个字段。

当某个方法通过 ART quick compiled code 路径执行时，运行时会取这个入口地址并跳过去。只要把 add 的入口改成自己的跳板，调用 add 时就可以先进入我们控制的代码。

1. 64 位下的字段偏移

示例的跳板代码硬编码了 entry_point_from_quick_compiled_code_ 的偏移：

// arm64: ldr x17, [x0, #32]
// x86_64: push [rdi + 0x20]

#32 和 0x20 都表示偏移 32 字节，也就是 0x20 。

按当前这个结构，在 64 位进程里可以这样理解：

偏移字段
0x00declaring_class_
0x04access_flags_
0x08dex_code_item_offset_
0x0cdex_method_index_
0x10method_index_
0x12hotness_count_
0x14padding
0x18ptr_sized_fields_.data_
0x20ptr_sized_fields_.entry_point_from_quick_compiled_code_

所以跳板里访问 [ArtMethod + 0x20] ，就是取替换方法自己的 quick entry。

注意：这是强版本相关的内部结构。不同 Android 版本、不同架构、不同 ART 实现细节都可能改变字段布局。写实验代码可以这样固定，写通用 Hook 框架就必须按 API 版本处理布局。

三、从 Method 对象拿到 ArtMethod 指针

Native 层第一步是找到 java.lang.reflect.Executable ：

jclass class_executable = env->FindClass("java/lang/reflect/Executable");

Method 和 Constructor 都继承自 Executable 。Android 的反射对象内部有一个隐藏字段：

long artMethod

示例通过 JNI 直接拿字段 ID：

jfieldID field_artMethod = env->GetFieldID(
    class_executable,
    "artMethod",
    "J"
);

这里的 "J" 是 JNI 签名，表示 Java long 。在 64 位进程里，它正好可以放下一个 native 指针。

接着找到当前 MainActivity 类中的两个方法：

jclass objClass = env->GetObjectClass(thiz);
auto method_add = env->GetMethodID(objClass, "add", "(II)I");
auto method_addHook = env->GetMethodID(objClass, "addHook", "(II)I");

(II)I 表示：

参数：int, int
返回：int

然后把 jmethodID 转成反射对象：

jobject obj_add = env->ToReflectedMethod(objClass, method_add, false);
jobject obj_addHook = env->ToReflectedMethod(objClass, method_addHook, false);

第三个参数 false 表示这是实例方法，不是静态方法。

最后读取 artMethod 字段：

ArtMethod *artMethod_add = reinterpret_cast<ArtMethod *>(
    env->GetLongField(obj_add, field_artMethod)
);

ArtMethod *artMethod_addHook = reinterpret_cast<ArtMethod *>(
    env->GetLongField(obj_addHook, field_artMethod)
);

到这里，代码已经从 Java 世界里的 Method 对象，跨到了 ART 内部的 ArtMethod* 。

可以把这段理解成：

jmethodID(add)
  ↓ ToReflectedMethod
java.lang.reflect.Method(add)
  ↓ GetLongField(artMethod)
ArtMethod* add

四、为什么不能直接把 add 的入口改成 addHook 的入口

一个很容易想到的写法是：

artMethod_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_ =
    artMethod_addHook->ptr_sized_fields_.entry_point_from_quick_compiled_code_;

但这个示例没有这么做，而是生成了一段 trampoline。

原因是 ART quick 调用约定里，进入 quick compiled code 时通常还需要一个 “当前方法” 的 ArtMethod* 。在这个示例覆盖的架构里：

架构方法寄存器
arm64x0
x86_64rdi

如果只把 add 的 entry 改成 addHook 的 entry，跳进 addHook 代码时，方法寄存器里可能仍然是 add 的 ArtMethod* 。这会让运行时栈回溯、异常处理、解释器桥接、元数据访问等行为出现不一致。

所以跳板要做的是：

method_register = artMethod_addHook;
jump(artMethod_addHook->entry_point_from_quick_compiled_code_);

也就是说，既要跳到 addHook 的代码入口，也要让 ART 认为当前执行的是 addHook 。

五、分配可执行跳板内存

示例先用 mmap 分配一页内存：

static void *allocTrampolineSpace() {
    unsigned char *buf = static_cast<unsigned char *>(mmap(
        NULL,
        TRAMPOLINE_SPACE_SIZE,
        PROT_READ | PROT_WRITE | PROT_EXEC,
        MAP_ANONYMOUS | MAP_PRIVATE,
        -1,
        0
    ));
    ...
    return buf;
}

这里的权限是：

权限含义
PROT_READ可读
PROT_WRITE可写，方便写入机器码
PROT_EXEC可执行，方便 CPU 执行这段机器码

示例代码为了简洁直接申请 RWX 内存。正式方案更常见的处理是：

  1. 先申请 RW 内存。
  2. 写入跳板机器码。
  3. 刷新 instruction cache。
  4. 再用 mprotect 改成 RX 。

这样可以减少可写可执行内存带来的安全风险，也更接近成熟 Hook 框架的做法。

六、arm64 跳板解析

arm64 分支的机器码如下：

unsigned char trampoline[] = {
        // OatQuickMethodHeader::code_size_.
        0x00, 0x00, 0x00, 0x00,
        // ldr x0, #12; ldr x17, [x0, #32]; br x17
        0x60, 0x00, 0x00, 0x58,
        0x11, 0x10, 0x40, 0xf9,
        0x20, 0x02, 0x1f, 0xd6,
        // ArtMethod* of addHook.
        0x00, 0x00, 0x00, 0x00,
        0x00, 0x00, 0x00, 0x00,
};
memcpy(trampoline + 16, &artMethod_addHook, sizeof(void *));

这段数据由三部分组成：

区域偏移作用
fake header0x00预留 OatQuickMethodHeader::code_size_
机器指令0x04真正执行的 trampoline
literal 数据0x10保存 artMethod_addHook 指针

写入内存后，代码会让入口地址跳过前 4 字节：

memcpy(targetAddr, trampoline, sizeof(trampoline));
targetAddr += 4;

所以 entry_point_from_quick_compiled_code_ 实际指向的是 trampoline + 4 ，也就是第一条 arm64 指令。

1. ldr x0, #12

第一条指令把后面的 literal 指针加载到 x0 ：

ldr x0, #12

当前指令地址是 trampoline + 4 ，向后 12 字节正好到 trampoline + 16 ，那里保存的是 artMethod_addHook 。

执行后：

x0 = artMethod_addHook

2. ldr x17, [x0, #32]

第二条指令从 addHook 的 ArtMethod 中读取 quick entry：

ldr x17, [x0, #32]

#32 就是 0x20 ，对应：

artMethod_addHook->ptr_sized_fields_.entry_point_from_quick_compiled_code_

执行后：

x17 = addHook.quick_entry

3. br x17

第三条指令跳到 addHook 的 quick entry：

br x17

最终效果：

add.quick_entry
  -> trampoline
  -> x0 = artMethod_addHook
  -> x17 = artMethod_addHook.quick_entry
  -> br x17
  -> addHook(...)

七、x86_64 跳板解析

x86_64 分支如下：

unsigned char trampoline[] = {
        // OatQuickMethodHeader::code_size_.
        0x00, 0x00, 0x00, 0x00,
        // mov rdi, ArtMethod*; push [rdi + 0x20]; ret
        0x48, 0xbf, 0x00, 0x00, 0x00, 0x00,
        0x00, 0x00, 0x00, 0x00,
        0xff, 0x77, 0x20,
        0xc3
};
memcpy(trampoline + 6, &artMethod_addHook, sizeof(void *));

同样，前 4 字节是 fake header，真正入口从 trampoline + 4 开始。

机器码对应的逻辑是：

mov rdi, artMethod_addHook
push qword ptr [rdi + 0x20]
ret

执行过程：

  1. mov rdi, artMethod_addHook ：把替换方法的 ArtMethod* 放进方法寄存器。
  2. push [rdi + 0x20] ：读取 addHook 的 quick entry，并压入栈顶。
  3. ret ：从栈顶弹出地址，相当于跳到 addHook.quick_entry 。

这里用 push + ret 做间接跳转，最终效果和 jmp [rdi + 0x20] 类似，但同时保留了 rdi = artMethod_addHook 。

八、替换 add 的 quick entry

跳板写入完成后，最关键的一行是：

*(uintptr_t **)&artMethod_add->ptr_sized_fields_.
        entry_point_from_quick_compiled_code_ = reinterpret_cast<uintptr_t *>(targetAddr);

如果写成更直观的伪代码，就是：

artMethod_add->entry_point_from_quick_compiled_code_ = trampoline_entry;

替换前：

add.ArtMethod.entry_point -> add compiled code

替换后：

add.ArtMethod.entry_point -> trampoline -> addHook compiled code

调用链可以画成这样：

重要的是：Dex 没有被改，Java 方法名也没有被改。 add 这个方法在反射、Dex 元数据、调用点上仍然叫 add ，只是它的运行入口被换掉了。

九、为什么 add 和 addHook 必须签名一致

本示例里两个方法签名完全一致：

add(int, int): int
addHook(int, int): int

这是必须的。

跳板只改了 “当前 ArtMethod 指针” 和 “下一跳地址”，没有重新打包参数，也没有做类型转换。原始调用 add(10, 20) 时，参数已经按照 add(int, int): int 的调用约定放在寄存器或栈上。跳到 addHook 后， addHook 会按自己的签名解释这些参数。

如果签名不一致，例如：

fun addHook(x: String): String

那么 addHook 会把原来的整数寄存器当成对象引用或其他类型解析，结果通常是崩溃、非法引用、返回值错乱。

所以 entrypoint 替换类 Hook 有一个基本前提：

替换方法必须能用同一套 ABI 参数布局接住原方法调用。

这也是成熟 Hook 框架要做大量桥接工作的原因。它们不能只支持 “同签名替换”，还要能进入统一回调、打包参数数组、调用 before/after、处理异常、再决定是否调用原方法。

十、access_flags_ 的处理

示例最后还改了 access_flags_ ：

uint32_t access_flags = artMethod_add->access_flags_;
access_flags |= 0x100;
artMethod_add->access_flags_ = access_flags;

0x100 对应 Java 方法标志里的 ACC_NATIVE 。

这一步要谨慎理解：真正让 add 改道的是 quick entry 被替换， ACC_NATIVE 不是这段跳转成立的唯一原因。把普通 Java 方法标成 native 会改变 ART 对方法类型的判断，可能影响反射、解释执行、栈回溯、调试和某些运行时分支。

实验代码里这样做，是为了让运行时在某些调用分支上更接近 “入口由 native/quick 桥接接管” 的状态。但如果要做通用方案，不能简单套一句 access_flags |= 0x100 ，而是要按 Android 版本和目标调用流程确认：

  • 是否真的需要设置 ACC_NATIVE
  • 是否需要保存原 flags 以便恢复
  • 是否会影响 Method.isNative()
  • 是否会影响解释器或 deopt 后的执行流程

更稳的做法是保存原始值：

auto old_flags = artMethod_add->access_flags_.load();
auto old_entry = artMethod_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_;

后续如果要卸载 Hook，可以把 flags 和 entry 都恢复。

十一、异常检查为什么重要

示例里有一个小工具：

static bool clearPendingException(JNIEnv *env, const char *message) {
    if (!env->ExceptionCheck()) {
        return false;
    }

    env->ExceptionDescribe();
    env->ExceptionClear();
    LOGE("%s", message);
    return true;
}

JNI 调用失败时，很多函数不会直接让 Native 崩掉，而是在线程上挂一个 pending exception。例如：

  • FindClass 找不到类
  • GetFieldID 找不到字段
  • GetMethodID 找不到方法
  • ToReflectedMethod 转换失败

如果 pending exception 不清理，后续 JNI 调用会继续处在异常状态，结果可能越来越怪。这里每一步失败都立刻打印、清理并返回 JNI_FALSE ，对调试 ART Hook 非常重要。

特别是这句：

env->GetFieldID(class_executable, "artMethod", "J");

它依赖 Android 内部实现。如果目标系统的 Executable 没有这个字段，或者字段布局变化，这一步就会失败。示例代码没有硬往下走，这是对的。

十二、当前示例的限制

这份代码非常适合学习，但它不是生产级 Hook 框架。主要限制如下。

1. 只支持 64 位 arm64 和 x86_64

代码显式排除了其他 ABI：

#if !defined(__aarch64__) && !defined(__x86_64__)
    LOGE("Unsupported ABI for trampoline");
    return JNI_FALSE;
#else
    ...
#endif

32 位 ARM、32 位 x86 的 ArtMethod 布局、寄存器约定、指令编码都不同，不能直接复用这两段跳板。

2. ArtMethod 布局强版本相关

[ArtMethod + 0x20] 是这份结构在 64 位下的假设。换一个 Android 版本后，字段顺序、压缩引用、对齐方式、quick entry 语义都可能变化。

更通用的方案至少要做到：

  • 按 Android API 版本选择结构定义
  • 对关键偏移做校验
  • 尽量从符号、源码或运行时行为推导偏移
  • 出错时停止安装 Hook，而不是继续写内存

3. 没有处理 instruction cache

代码把机器码写入可执行内存后，直接让 CPU 去执行。更完整的写法应该在写入后刷新指令缓存：

__builtin___clear_cache(
    reinterpret_cast<char *>(trampoline_start),
    reinterpret_cast<char *>(trampoline_start + trampoline_size)
);

在 ARM 架构上，动态生成代码后刷新 instruction cache 是很重要的。否则数据已经写进内存，但指令流水线可能仍然看不到最新内容。

4. 没有处理并发调用

示例是在点击按钮时安装 Hook，而且方法非常简单。如果目标方法正在其他线程执行，同时修改它的 entry，就可能出现竞态。

成熟方案通常要考虑：

  • 尽量在方法首次调用前安装 Hook
  • 暂停相关线程或使用运行时安全点
  • 原子地切换入口
  • 保存并恢复旧入口

5. 可能被 JIT/AOT、内联影响

修改 ArtMethod 的 quick entry，并不等于所有调用点都会重新走这个入口。

如果某些调用已经被编译器内联：

val r = 10 + 20

那就根本不会再读取 add.ArtMethod.entry_point 。如果某个调用点已经被 AOT/JIT 优化成直接分支，也可能绕过你后面改的入口。

所以真实框架里经常还要配合：

  • deopt
  • 禁止内联
  • 清理 inline cache
  • 让目标方法走解释器或统一桥

6. 没有 “调用原方法” 的能力

当前示例是硬替换：

add -> addHook

它没有 before/after 回调，也没有保存原始入口后再调用原方法。

如果要做成类似 Xposed 的能力，需要额外设计：

  • 原始 entry 备份
  • replacement callback
  • 参数打包和解包
  • 返回值修改
  • 异常透传
  • 调原方法时避免再次进入 Hook

十三、和 Xposed Hook 的关系

传统 Xposed 写法是：

XposedHelpers.findAndHookMethod(
    Target.class,
    "add",
    int.class,
    int.class,
    new XC_MethodHook() {
        @Override
        protected void afterHookedMethod(MethodHookParam param) {
            param.setResult(200);
        }
    }
);

使用者看到的是 “找到方法、注册回调、修改结果”。但框架底层必须解决的问题和本文类似：

  1. 如何定位目标方法的运行时表示。
  2. 如何让目标方法进入框架控制的入口。
  3. 如何保存原始调用能力。
  4. 如何处理解释执行、编译执行、JIT、AOT、内联。
  5. 如何跨 Android 版本兼容 ART 内部结构。

本文这个示例只做了其中最小的一段：

定位 ArtMethod -> 替换 quick entry -> 跳到同签名方法

它没有 Xposed 的回调链，也没有通用参数处理，但正因为简单，反而能看清楚方法入口替换的骨架。

十四、可以怎样继续扩展

如果基于这份代码继续做实验，可以按下面几个方向加功能。

1. 打印 ArtMethod 关键字段

安装 Hook 前后打印：

LOGD("add ArtMethod: %p", artMethod_add);
LOGD("addHook ArtMethod: %p", artMethod_addHook);
LOGD("add entry: %p", artMethod_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_);
LOGD("addHook entry: %p", artMethod_addHook->ptr_sized_fields_.entry_point_from_quick_compiled_code_);
LOGD("add flags: 0x%x", artMethod_add->access_flags_.load());

这样可以直观看到 entry 是否真的被改掉。

2. 做一个 unhook

安装时保存：

static void *origin_entry = nullptr;
static uint32_t origin_flags = 0;

卸载时恢复：

artMethod_add->ptr_sized_fields_.entry_point_from_quick_compiled_code_ = origin_entry;
artMethod_add->access_flags_ = origin_flags;

这能帮助理解 “Hook 本质上是状态替换”，也方便反复测试。

3. 支持调用原方法

当前 add 只能彻底进入 addHook 。如果想在 addHook 里调用原 add ，就不能简单调用 add() ，否则又会进入 Hook。

常见思路是：

  • 保存原始 quick entry
  • 构造一个 origin trampoline
  • 调用原始 entry 时临时恢复或走备份入口
  • 注意避免递归进入 Hook

这一步开始就接近真正 Hook 框架的复杂区了。

4. 做版本适配表

可以为不同 Android 版本维护：

struct ArtMethodOffsets {
    size_t access_flags;
    size_t entry_point_from_quick_compiled_code;
};

然后按 API 版本选择偏移，而不是把 0x20 写死在所有逻辑里。

十五、总结

这份示例可以用一句话概括：

从 Java Method 取出 ArtMethod*，把目标方法的 quick entry 改成自定义跳板，跳板再带着替换方法的 ArtMethod* 进入替换方法 quick entry。

最关键的几个点：

  • Executable.artMethod 是 Java 反射对象通往 ART 内部方法对象的入口。
  • entry_point_from_quick_compiled_code_ 决定 quick compiled code 路径下方法从哪里开始执行。
  • trampoline 不能只跳地址，还要修正当前 ArtMethod* 所在寄存器。
  • 同签名替换能成立，是因为原始参数布局可以直接复用。
  • ArtMethod Hook 强依赖 Android 版本、ABI、编译状态和运行时细节，实验时很爽，工程化时很考验耐心。

读懂这一层以后，再看 Xposed、LSPosed 或其他 ART Hook 框架，就不再只是 “框架帮我 Hook 了一个方法”，而能看到背后真正被改写的运行时入口。
