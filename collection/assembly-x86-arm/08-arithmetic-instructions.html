算术指令

以下对比 x86-64 Linux、AArch64 Linux、macOS arm64 三种环境里的整数算术指令。

汇编里的算术指令做的事情很直接：

把寄存器或内存里的整数拿来做加、减、乘、除、取负、带进位运算

但不同架构的写法和副作用差异很大：

  • x86-64 很多算术指令会自动更新 RFLAGS 。
  • AArch64 普通 add/sub 默认不更新标志位，要用 adds/subs 才更新 NZCV 。
  • x86-64 的除法使用隐式寄存器组合，例如 %rdx:%rax 。
  • AArch64 的除法把商放进普通寄存器，余数需要手动用 msub 算出来。

说明：AArch64 和 ARM64 在这里都指 64 位 Arm 架构。Linux AArch64 和 macOS arm64 使用相同的核心算术指令，但入口、链接方式、系统调用方式和全局符号写法不同。

两个最重要的顺序差异：

# x86-64 AT&T：rax = rax + 5
add $5, %rax

// AArch64：x0 = x0 + 5
add x0, x0, #5

x86-64 的目标寄存器通常在最后；AArch64 的目标寄存器通常在最前。

2. 加法和减法

2.1 x86-64

mov $7, %rax        # rax = 7
add $5, %rax        # rax = rax + 5 = 12
sub $2, %rax        # rax = rax - 2 = 10

add 和 sub 会更新 RFLAGS ，例如：

  • ZF ：结果是否为 0。
  • SF ：结果最高位是否为 1，常用于有符号数判断。
  • CF ：无符号运算是否进位或借位。
  • OF ：有符号运算是否溢出。

2.2 AArch64/macOS arm64

mov x0, #7          // x0 = 7
add x0, x0, #5      // x0 = x0 + 5 = 12
sub x0, x0, #2      // x0 = x0 - 2 = 10

普通 add/sub 不更新 NZCV 。如果后面要根据结果跳转，要用 adds/subs 或 cmp ：

adds x0, x0, #5     // x0 = x0 + 5，并更新 NZCV
subs x0, x0, #2     // x0 = x0 - 2，并更新 NZCV
cmp x0, #10         // 只比较，不保存结果；本质类似 subs xzr, x0, #10

3. 三个平台示例：加、减、乘

目标计算：

(7 + 5) * 3 - 6 = 30

3.1 x86-64 Linux

文件名： arith-x86-64.s

.section .text
.global _start
_start:
    # 目标：计算 (7 + 5) * 3 - 6 = 30，然后把 30 作为退出码。
    #
    # AT&T 语法：
    # add $5, %rax  表示 rax = rax + 5
    # sub $6, %rax  表示 rax = rax - 6

    mov $7, %rax                            # rax = 7
    add $5, %rax                            # rax = 12

    # imul 有多种形式。这里使用三操作数形式：
    # imul $3, %rax, %rax 表示 rax = rax * 3。
    imul $3, %rax, %rax                     # rax = 36

    sub $6, %rax                            # rax = 30

    # exit(rax)
    mov %rax, %rdi                          # rdi = 30，退出码
    mov $60, %rax                           # rax = 60，Linux x86-64 exit 系统调用号
    syscall

编译运行：

as --64 arith-x86-64.s -o arith-x86-64.o
ld arith-x86-64.o -o arith-x86-64
./arith-x86-64
echo $?

预期输出：

30

3.2 AArch64 Linux

文件名： arith-aarch64.s

.section .text
.global _start
_start:
    // 目标：计算 (7 + 5) * 3 - 6 = 30，然后把 30 作为退出码。

    mov x0, #7                              // x0 = 7
    add x0, x0, #5                          // x0 = 12

    mov x1, #3                              // x1 = 3
    mul x0, x0, x1                          // x0 = x0 * x1 = 36

    sub x0, x0, #6                          // x0 = 30

    // exit(x0)
    mov x8, #93                             // x8 = 93，Linux AArch64 exit 系统调用号
    svc #0

编译运行：

as arith-aarch64.s -o arith-aarch64.o
ld arith-aarch64.o -o arith-aarch64
./arith-aarch64
echo $?

预期输出：

30

3.3 macOS arm64

文件名： arith-macos-arm64.s

.text
.globl _main
.p2align 2
_main:
    // 目标：计算 (7 + 5) * 3 - 6 = 30，然后让 main 返回 30。

    mov x0, #7                              // x0 = 7
    add x0, x0, #5                          // x0 = 12

    mov x1, #3                              // x1 = 3
    mul x0, x0, x1                          // x0 = x0 * x1 = 36

    sub x0, x0, #6                          // x0 = 30

    // main 的返回值放在 w0/x0。
    // C 运行时会把低 8 位作为进程退出码，所以 echo $? 会看到 30。
    ret

编译运行：

clang arith-macos-arm64.s -o arith-macos-arm64
./arith-macos-arm64
echo $?

预期输出：

30

4. 乘法： imul/mul 和 mul/madd

4.1 x86-64 的 imul

imul 是有符号整数乘法，常见几种形式：

imul %rbx              # 隐式：rdx:rax = rax * rbx
imul %rbx, %rax        # rax = rax * rbx
imul $3, %rax, %rax    # rax = rax * 3

入门阶段最推荐看二操作数和三操作数形式，因为目标寄存器清楚：

mov $12, %rax
imul $3, %rax, %rax    # rax = 36

无符号乘法是 mul ，通常使用隐式寄存器，结果可能放进 %rdx:%rax 。初学时如果只是做普通小整数乘法，先用 imul 的二 / 三操作数形式更好懂。

4.2 AArch64 的 mul 、 madd 、 msub

AArch64 的乘法写法更规整：

mul x0, x1, x2         // x0 = x1 * x2

还有很实用的乘加和乘减：

madd x0, x1, x2, x3    // x0 = x1 * x2 + x3
msub x0, x1, x2, x3    // x0 = x3 - x1 * x2

madd 很适合写：

a * b + c

msub 很适合配合除法算余数：

remainder = dividend - quotient * divisor

5. 除法：x86-64 隐式寄存器，AArch64 显式寄存器

5.1 x86-64 有符号除法 idiv

x86-64 的 idiv 比加减乘更容易让人愣住，因为使用隐式寄存器：

被除数：rdx:rax 组成的 128 位值
除数：idiv 后面的操作数
商：rax
余数：rdx

做 64 位有符号除法前，常用 cqto 把 %rax 符号扩展到 %rdx:%rax ：

mov $100, %rax
cqto
mov $7, %rbx
idiv %rbx          # rax = 14, rdx = 2

如果忘记准备 %rdx:%rax ，除法结果可能错，严重时还会触发除法异常。

5.2 AArch64 有符号除法 sdiv

AArch64 的除法更直观：

mov x1, #100
mov x2, #7
sdiv x0, x1, x2    // x0 = x1 / x2 = 14

但 sdiv 只给商，不给余数。余数常用 msub 计算：

msub x3, x0, x2, x1 // x3 = x1 - x0 * x2 = 2

6. 三个平台示例：除法和余数

目标计算：

100 / 7 = 14 余 2
返回 14 + 2 = 16

6.1 x86-64 Linux

文件名： div-x86-64.s

.section .text
.global _start
_start:
    # 目标：计算 100 / 7，得到商 14、余数 2，
    #      再返回 14 + 2 = 16。

    mov $100, %rax                           # rax = 被除数低 64 位
    cqto                                     # 把 rax 符号扩展到 rdx:rax

    mov $7, %rbx                             # rbx = 除数
    idiv %rbx                                # rax = 商 14；rdx = 余数 2

    add %rdx, %rax                           # rax = 商 + 余数 = 16

    # exit(rax)
    mov %rax, %rdi
    mov $60, %rax
    syscall

运行：

as --64 div-x86-64.s -o div-x86-64.o
ld div-x86-64.o -o div-x86-64
./div-x86-64
echo $?

预期输出：

16

6.2 AArch64 Linux

文件名： div-aarch64.s

.section .text
.global _start
_start:
    // 目标：计算 100 / 7，得到商 14、余数 2，
    //      再返回 14 + 2 = 16。

    mov x1, #100                             // x1 = 被除数 dividend
    mov x2, #7                               // x2 = 除数 divisor

    sdiv x0, x1, x2                          // x0 = 商 quotient = x1 / x2 = 14
    msub x3, x0, x2, x1                      // x3 = x1 - x0 * x2 = 余数 2

    add x0, x0, x3                           // x0 = 商 + 余数 = 16

    // exit(x0)
    mov x8, #93
    svc #0

运行：

as div-aarch64.s -o div-aarch64.o
ld div-aarch64.o -o div-aarch64
./div-aarch64
echo $?

预期输出：

16

6.3 macOS arm64

文件名： div-macos-arm64.s

.text
.globl _main
.p2align 2
_main:
    // 目标：计算 100 / 7，得到商 14、余数 2，
    //      再让 main 返回 14 + 2 = 16。

    mov x1, #100                             // x1 = 被除数 dividend
    mov x2, #7                               // x2 = 除数 divisor

    sdiv x0, x1, x2                          // x0 = 商 quotient = 14
    msub x3, x0, x2, x1                      // x3 = 余数 remainder = 2

    add x0, x0, x3                           // x0 = 16
    ret

运行：

clang div-macos-arm64.s -o div-macos-arm64
./div-macos-arm64
echo $?

预期输出：

16

7. 有符号和无符号

CPU 寄存器里只有位模式，本身不带 “有符号” 或 “无符号” 标签。同一串位可以按两种方式解释：

0xffffffffffffffff
无符号解释：18446744073709551615
有符号解释：-1

所以很多指令分有符号和无符号版本：

操作x86-64 有符号x86-64 无符号AArch64 有符号AArch64 无符号
乘法imulmulmul 低位结果相同mul 低位结果相同
高半部分乘法imul 隐式形式mul 隐式形式smulhumulh
除法idivdivsdivudiv
比较后跳转jl/jg 等jb/ja 等b.lt/b.gt 等b.lo/b.hi 等

加法和减法的位级计算一样，但解释标志位时要分清：

  • 无符号溢出看 CF / C 。
  • 有符号溢出看 OF / V 。

8. 标志位：x86-64 默认更新，AArch64 要选带 s

8.1 x86-64 的 RFLAGS

x86-64 的很多算术指令都会更新标志位：

mov $255, %al
add $1, %al        # al = 0；CF = 1；ZF = 1

解释：

  • %al 只有 8 位， 255 + 1 会绕回 0 。
  • 对无符号 8 位数来说，这发生了进位，所以 CF = 1 。
  • 结果是 0，所以 ZF = 1 。

常见标志：

标志含义常见用途
ZFzero flag，结果是否为 0相等判断
SFsign flag，结果最高位是否为 1有符号正负
CFcarry flag，无符号进位 / 借位无符号大小、宽整数运算
OFoverflow flag，有符号溢出有符号溢出判断

8.2 AArch64 的 NZCV

AArch64 普通 add/sub 不更新标志：

add x0, x0, #1      // 不更新 NZCV
sub x0, x0, #1      // 不更新 NZCV

要更新标志，用带 s 的版本：

adds x0, x0, #1     // 更新 NZCV
subs x0, x0, #1     // 更新 NZCV
cmp x0, #10         // 比较，更新 NZCV，不保存结果

NZCV 含义：

标志含义对应 x86-64 直觉
Nnegative，结果最高位为 1类似 SF
Zzero，结果为 0类似 ZF
Ccarry，无符号进位；减法时表示没有借位类似 CF ，但减法语义要小心
Voverflow，有符号溢出类似 OF

9. 带进位和借位： adc/sbb 与 adcs/sbcs

需要计算超过一个寄存器宽度的大整数时，需要把低位加法产生的进位传给高位。

x86-64：

# 计算 128 位加法：
# (rdx:rax) = (rdx:rax) + (rcx:rbx)
add %rbx, %rax      # 低 64 位相加，产生的进位放入 CF
adc %rcx, %rdx      # 高 64 位相加，再加上 CF

AArch64：

// 计算 128 位加法：
// (x1:x0) = (x1:x0) + (x3:x2)
adds x0, x0, x2     // 低 64 位相加，更新 C
adcs x1, x1, x3     // 高 64 位相加，再加上 C

减法也类似：

x86-64：

sub %rbx, %rax      # 低 64 位相减
sbb %rcx, %rdx      # 高 64 位相减，再减去借位

AArch64：

subs x0, x0, x2
sbcs x1, x1, x3

这个主题会在后面的多精度整数、加密和大数处理里经常出现。入门阶段先记住：低位先用更新标志的加减，高位再用带 carry/borrow 的指令。

10. 自增、自减和取负

x86-64：

inc %rax            # rax = rax + 1
dec %rax            # rax = rax - 1
neg %rax            # rax = -rax

注意： inc/dec 不更新 CF ，这和 add $1 / sub $1 不完全一样。现代编译器很多时候更偏向用 add/sub ，因为标志位行为更直观。

AArch64/macOS arm64 没有专门的通用 inc/dec 指令，常用：

add x0, x0, #1      // x0++
sub x0, x0, #1      // x0--
neg x0, x0          // x0 = -x0

如果要更新标志，用：

adds x0, x0, #1
subs x0, x0, #1
negs x0, x0

11. 算术和内存

x86-64 可以直接对内存做一些算术：

addq $1, counter(%rip)       # counter = counter + 1
subq $2, counter(%rip)       # counter = counter - 2

这很方便，但要记住会读内存、计算、再写回内存。

AArch64 是 load/store 架构，不能直接这样写：

add [x9], [x9], #1           // 错误示意

要分三步：

ldr x0, [x9]                 // 读变量
add x0, x0, #1               // 算术
str x0, [x9]                 // 写回变量

macOS arm64 同样遵守 AArch64 这套 load/store 风格。

12. 除法异常和边界

除法比加减乘更危险，至少要注意：

  • 除数不能是 0。
  • x86-64 的 idiv/div 如果商放不进目标寄存器，会触发除法异常。
  • 有符号最小值除以 -1 会溢出，例如 64 位 INT64_MIN / -1 。
  • AArch64 的 sdiv/udiv 对除以 0 不会像 x86 那样直接触发同类除法异常；架构规定结果为 0，但实际程序仍然应该避免除以 0。

入门示例都选择了安全数字，例如 100 / 7 。

真实程序里，在除法前先检查除数：

# x86-64 思路
cmp $0, %rbx
je divide_by_zero
// AArch64 思路
cbz x2, divide_by_zero

13. 常见坑

13.1 x86-64 AT&T 操作数顺序写反

sub $3, %rax        # rax = rax - 3
sub %rax, $3        # 错误，立即数不能当目标

再看寄存器版本：

sub %rbx, %rax      # rax = rax - rbx

不是 %rbx = %rbx - %rax 。

13.2 AArch64 忘记目标寄存器

AArch64 通常要写目标寄存器：

add x0, x0, x1      // x0 = x0 + x1

不要按 x86 直觉写成：

add x1, x0          // 错误示意

13.3 AArch64 用 add 后期待标志位更新

add x0, x0, #1      // 不更新 NZCV
b.eq label          // 这里不会基于这条 add 的结果判断

要写：

adds x0, x0, #1     // 更新 NZCV
b.eq label

或者：

add x0, x0, #1
cmp x0, #0
b.eq label

13.4 x86-64 除法前忘记 cqto

mov $100, %rax
mov $7, %rbx
idiv %rbx           # 错误风险：rdx 没准备好

正确：

mov $100, %rax
cqto
mov $7, %rbx
idiv %rbx

13.5 把退出码当完整整数输出

echo $? 通常只显示进程退出状态的低 8 位。比如返回 300 ，可能看到 44 ，因为：

300 mod 256 = 44

所以示例都选择 0 到 255 之间的结果。

14. 小练习

  1. 把 (7 + 5) * 3 - 6 改成 (9 - 4) * 6 + 2 ，让退出码变成 32 。
  2. 把除法示例改成 123 / 10 ，返回商加余数，应该是 12 + 3 = 15 。
  3. 在 AArch64 示例里把 add 改成 adds ，再尝试接 b.eq 做条件跳转。
  4. 在 x86-64 里写一个 128 位加法片段，使用 add + adc 。
  5. 故意在 x86-64 除法前删掉 cqto ，观察程序是否还能稳定得到正确结果，并思考为什么这是危险写法。
